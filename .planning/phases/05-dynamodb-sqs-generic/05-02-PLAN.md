    ```ts
    export const AUTHORIZATION_DDB =
      'AWS4-HMAC-SHA256 Credential=test/20260417/us-east-1/dynamodb/aws4_request'

    const DDB_TARGET_PREFIX = 'DynamoDB_20120810'

    export async function ddbJsonCall<T = unknown>(action: string, body: object): Promise<T> {
      const res = await fetch('/', {
        method: 'POST',
        headers: {
          'Content-Type': 'application/x-amz-json-1.0',
          // Pitfall 7.2.1 / §1: dispatch is by HEADER, not URL path.
          'X-Amz-Target': `${DDB_TARGET_PREFIX}.${action}`,
          Authorization: AUTHORIZATION_DDB,
        },
        body: JSON.stringify(body),
      })
      if (!res.ok) {
        const text = await res.text()
        // Backend errors look like { __type: 'ResourceNotFoundException', message: '...' }
        throw new Error(`DynamoDB ${action} failed ${res.status}: ${text}`)
      }
      return res.json() as Promise<T>
    }
    ```

    ```ts
    import type { DdbAttributeValue, DdbScalarType } from '../../../shared/types'

    export const SCALAR_TYPES: readonly DdbScalarType[] = ['S', 'N', 'B', 'BOOL', 'NULL'] as const

    export function isScalarType(t: string): t is DdbScalarType {
      return (SCALAR_TYPES as readonly string[]).includes(t)
    }

    // Thrown when caller passes a complex type (L/M/SS/NS/BS) to the scalar codec.
    // The ItemForm (D-03/D-06) catches this and shows a "use JSON mode" hint.
    export class DdbComplexTypeNotSupportedError extends Error {
      constructor(type: string) {
        super(`AttributeValue type "${type}" requires JSON advanced mode (D-06 — scalar-only form)`)
        this.name = 'DdbComplexTypeNotSupportedError'
      }
    }

    export function marshalScalar(
      value: string | number | boolean | null,
      type: DdbScalarType,
    ): DdbAttributeValue {
      switch (type) {
        case 'S':    return { S: String(value ?? '') }
        // Pitfall 7.2.1: N is ALWAYS string on the wire, even when input is a number.
        case 'N':    return { N: String(value ?? '0') }
        case 'BOOL': return { BOOL: Boolean(value) }
        // NULL is a marker — any input value is discarded.
        case 'NULL': return { NULL: true }
        // B expects pre-encoded base64 string from the File upload widget.
        case 'B':    return { B: String(value ?? '') }
      }
    }

    export function unmarshalScalar(av: DdbAttributeValue): string | number | boolean | null {
      if ('S' in av)    return av.S
      if ('N' in av)    return av.N                      // preserve string — caller decides Number() for display
      if ('BOOL' in av) return av.BOOL
      if ('NULL' in av) return null
      if ('B' in av)    return av.B
      // Any remaining key is a complex type.
      const type = Object.keys(av)[0] ?? 'unknown'
      throw new DdbComplexTypeNotSupportedError(type)
    }

    // Display-only flattener — used by the items Table cell renderer.
    // Handles ALL types (scalar + complex) because read-side may legitimately surface complex values.
    export function renderAttributeValue(av: DdbAttributeValue): string {
      if ('S' in av)    return av.S
      if ('N' in av)    return av.N
      if ('BOOL' in av) return av.BOOL ? 'true' : 'false'
      if ('NULL' in av) return '(null)'
      if ('B' in av)    return `(binary, ${av.B.length} b64 chars)`
      if ('L' in av)    return `[L: ${av.L.length} items]`
      if ('M' in av)    return `[M: ${Object.keys(av.M).length} keys]`
      if ('SS' in av)   return `[SS: ${av.SS.length} strings]`
      if ('NS' in av)   return `[NS: ${av.NS.length} numbers]`
      if ('BS' in av)   return `[BS: ${av.BS.length} blobs]`
      return JSON.stringify(av)
    }
    ```
