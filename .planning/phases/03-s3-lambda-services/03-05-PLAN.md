    ```tsx
    type DropZoneProps = {
      onDrop: (files: File[]) => void
      disabled?: boolean
      currentPrefix: string
      children: ReactNode
    }
    export function DropZone({ onDrop, disabled, currentPrefix, children }: DropZoneProps) {
      const [dragDepth, setDragDepth] = useState(0)
      const over = dragDepth > 0
      return (
        <div
          role="region"
          aria-label={copy.s3.dropZoneAria}
          aria-live="polite"
          aria-atomic="true"
          data-drag-over={over ? 'true' : 'false'}
          className={over ? 's3-dropzone s3-dropzone--over' : 's3-dropzone'}
          onDragEnter={(e) => { e.preventDefault(); if (!disabled) setDragDepth((d) => d + 1) }}
          onDragOver={(e) => e.preventDefault()}
          onDragLeave={() => setDragDepth((d) => Math.max(0, d - 1))}
          onDrop={(e) => {
            e.preventDefault()
            setDragDepth(0)
            if (disabled) return
            const files = Array.from(e.dataTransfer.files || [])
            if (files.length) onDrop(files)
          }}
        >
          {over && (
            <div className="s3-dropzone__overlay" role="status">
              {currentPrefix ? copy.s3.dropOverlayWithPrefix(currentPrefix) : copy.s3.dropOverlayRoot}
            </div>
          )}
          {children}
        </div>
      )
    }
    ```
