    ```tsx
    export default function ObjectBrowserPage() {
      const { bucketName = '' } = useParams<{ bucketName: string }>()
      const [searchParams, setSearchParams] = useSearchParams()
      const prefix = searchParams.get('prefix') ?? ''
      const pageSize = useUiStore((s) => s.s3ObjectPageSize)
      const [tokens, setTokens] = useState<string[]>([])
      const [currentPage, setCurrentPage] = useState(0)
      const [selected, setSelected] = useState<S3ObjectEntry[]>([])

      // Pitfall 3: reset on prefix change
      useEffect(() => { setTokens([]); setCurrentPage(0); setSelected([]) }, [prefix])

      const currentToken = currentPage === 0 ? null : (tokens[currentPage - 1] ?? null)
      const { data, isLoading, error, refetch } = useObjects({ bucket: bucketName, prefix, pageSize, continuationToken: currentToken })

      const onNext = () => {
        if (data?.nextContinuationToken) {
          setTokens((prev) => [...prev.slice(0, currentPage), data.nextContinuationToken!])
          setCurrentPage((p) => p + 1)
        }
      }
      const onPrev = () => {
        if (currentPage > 0) setCurrentPage((p) => p - 1)
      }
      const canNext = Boolean(data?.isTruncated)  // Pitfall 4
      const canPrev = currentPage > 0

      const onFolderClick = (folderKey: string) => setSearchParams({ prefix: prefix + folderKey.slice(prefix.length) })
      // Simpler: onFolderClick receives the FULL CommonPrefix key (e.g. 'photos/2026/'), just setSearchParams({prefix: folderKey}).
      // Use whichever yields correct navigation; tests are the gate.

      const onParentClick = () => {
        const parts = prefix.replace(/\/$/, '').split('/').filter(Boolean)
        parts.pop()
        const next = parts.length ? parts.join('/') + '/' : ''
        if (next) setSearchParams({ prefix: next }); else setSearchParams({})
      }

      const onBreadcrumbNavigate = (newPrefix: string) => {
        if (newPrefix) setSearchParams({ prefix: newPrefix }); else setSearchParams({})
      }

      // Plan 05 will replace this placeholder with the real ObjectDetail:
      const { setContent, setOpen } = useSplitPanel()
      const onFileClick = (entry: Extract<S3ObjectEntry,{kind:'file'}>) => {
        setContent(<Box padding="m">{entry.key} (detail view wired in Plan 05)</Box>)
        setOpen(true)
      }

      return (
        <SpaceBetween size="m">
          <Header variant="h1" actions={<CopyToClipboard textToCopy={bucketName} copyButtonText={copy.s3.copyBucketNameButton}/>}>
            {bucketName}
          </Header>
          <PrefixBreadcrumb bucketName={bucketName} prefix={prefix} onNavigate={onBreadcrumbNavigate}/>
          <ObjectTable
            bucket={bucketName}
            prefix={prefix}
            data={data}
            isLoading={isLoading}
            error={error as Error | null}
            selected={selected}
            onSelectionChange={setSelected}
            onFileClick={onFileClick}
            onFolderClick={onFolderClick}
            onParentClick={onParentClick}
            onRefresh={() => refetch()}
            onUploadClick={() => { /* TODO Plan 05: wire upload + delete */ }}
            onDeleteSelected={() => { /* TODO Plan 05 */ }}
            pagination={{ canPrev, canNext, onPrev, onNext, pageLabel: `Page ${currentPage + 1}` }}
          />
        </SpaceBetween>
      )
    }
    ```
