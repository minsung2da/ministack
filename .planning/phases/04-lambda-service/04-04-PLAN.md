    import { useEffect, useState } from 'react'
    import { useParams, Link } from 'react-router-dom'
    import { Tabs, Header, Alert, Box, Spinner, BreadcrumbGroup, CopyToClipboard, SpaceBetween } from '@cloudscape-design/components'
    import { useFunction } from './api/useFunction'
    import { ConfigurationPanel } from './components/ConfigurationPanel'
    import { EnvironmentPanel } from './components/EnvironmentPanel'
    import { TriggersPanel } from './components/TriggersPanel'
    import { copy } from '../../shared/copy'

    export default function FunctionDetailPage() {
      const { functionName = '' } = useParams<{ functionName: string }>()
      const { data, isLoading, error } = useFunction(functionName)
      const [activeTabId, setActiveTabId] = useState<string>('configuration')
      const [payload, setPayload] = useState<string>('{}')

      // Pitfall 10 + Open Question 3: reset payload when function name changes
      useEffect(() => { setPayload('{}') }, [functionName])

      if (isLoading) return <Spinner />

          {isNotFound

            : (error as Error).message}
        </Alert>
      }
      if (!data) return null

      return <SpaceBetween size="l">
        <BreadcrumbGroup items={[
          { text: 'Console', href: '/_console' },
          { text: 'Lambda', href: '/_console/services/lambda' },
          { text: functionName, href: '' },

                {/* TODO Plan 05: replace with <InvokePanel functionName={functionName} payload={payload} onPayloadChange={setPayload} /> */}
              </Box> },

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| URL `:functionName` route param → useFunction query | User-controlled string becomes server query |
| Env var values → rendered text | Emulator-sourced strings rendered to the user (plaintext per D-05) |
| External Function URL href → browser navigation | User could click through to an external URL |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-4-04-01 | Tampering (path traversal in functionName) | useFunction path builder | mitigate | `encodeURIComponent(name)` at the hook layer (Plan 02). Backend `_resolve_name` treats name as opaque lookup key. |
| T-4-04-02 | Information Disclosure (env values in plaintext) | EnvironmentPanel | accept | D-05 explicit: local emulator, no shoulder-surfing model. Documented. |
| T-4-04-03 | XSS via env var values | EnvironmentPanel cell render | mitigate | Rendered via React `{e.value}` (auto-escaped). No dangerouslySetInnerHTML. Test greps for absence of masking artifacts also covers this indirectly. |
| T-4-04-04 | Tampering (malicious FunctionUrl redirect) | TriggersPanel Link | mitigate | `<Link href={u.FunctionUrl} external>` — Cloudscape adds `rel="noopener noreferrer"` and target="_blank" for external links. URL sourced from backend, not user input. |
| T-4-04-05 | Business Logic (V11) accidental trigger mutation | TriggersPanel | mitigate | No mutation controls present (D-07). Grep check in acceptance_criteria enforces. |
| T-4-04-06 | State leakage (payload cross-function) | FunctionDetailPage useEffect reset | mitigate | `useEffect(() => setPayload('{}'), [functionName])` — test covers. |
</threat_model>
