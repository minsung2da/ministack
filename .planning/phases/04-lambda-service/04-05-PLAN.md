    ```tsx
    type Props = {
      payload: string
      onPayloadChange: (s: string) => void
      onValidityChange?: (valid: boolean) => void
      disabled?: boolean
    }

    export function PayloadEditor({ payload, onPayloadChange, onValidityChange, disabled }: Props) {
      const [sampleId, setSampleId] = useState<string | null>(null)
      const validation = validateJson(payload)
      // Report validity upward (useEffect so we don't thrash during render)
      useEffect(() => { onValidityChange?.(validation.valid) }, [validation.valid, onValidityChange])

      return <SpaceBetween size="s">
        <FormField label={copy.lambda.payloadLabel} errorText={!validation.valid ? copy.lambda.payloadInvalidJson(validation.error ?? '') : undefined}>
          <Textarea
            value={payload}
            onChange={(e) => onPayloadChange(e.detail.value)}
            rows={12}
            spellcheck={false}
            disabled={disabled}
          />
        </FormField>
        <FormField label={copy.lambda.payloadSampleLabel}>
          <Select
            selectedOption={sampleId ? SAMPLE_PAYLOADS.find((s) => s.label === sampleId).map… /* pick the Option shape */ : null}
            onChange={(e) => {
              const chosen = SAMPLE_PAYLOADS.find((s) => s.label === e.detail.selectedOption?.value)
              if (chosen) { onPayloadChange(chosen.value); setSampleId(chosen.label) }
            }}
            options={SAMPLE_PAYLOADS.map((s) => ({ label: s.label, value: s.label }))}
            placeholder="Load sample payload"
          />
        </FormField>
      </SpaceBetween>
    }

    function validateJson(s: string): { valid: boolean; error?: string } {
      try { JSON.parse(s); return { valid: true } } catch (e) { return { valid: false, error: (e as Error).message } }
    }
    ```

    ```tsx
    type Props = { functionName: string; payload: string; onPayloadChange: (s: string) => void }

    export function InvokePanel({ functionName, payload, onPayloadChange }: Props) {
      const [isValid, setIsValid] = useState(true)
      const [spinnerCopy, setSpinnerCopy] = useState(copy.lambda.invokingCopy)
      const mutation = useInvoke(functionName)

      // D-09 copy swap at 3 seconds
      useEffect(() => {
        if (!mutation.isPending) { setSpinnerCopy(copy.lambda.invokingCopy); return }
        const timer = setTimeout(() => setSpinnerCopy(copy.lambda.stillInvokingCopy), 3000)
        return () => clearTimeout(timer)
      }, [mutation.isPending])

      const onSubmit = () => {
        if (!isValid) return
        try {
          const parsed = JSON.parse(payload)
          mutation.mutate(parsed)
        } catch {
          // isValid guard should have prevented; no-op
        }
      }

      return <SpaceBetween size="l">
        <PayloadEditor payload={payload} onPayloadChange={onPayloadChange} onValidityChange={setIsValid} disabled={mutation.isPending} />
        <SpaceBetween size="xs" direction="horizontal">
          <Button variant="primary" onClick={onSubmit} disabled={!isValid || mutation.isPending || !functionName}>
            {copy.lambda.invokeButton}
          </Button>
          {mutation.isPending && <SpaceBetween size="xs" direction="horizontal"><Spinner /><Box variant="span">{spinnerCopy}</Box></SpaceBetween>}
        </SpaceBetween>
        {mutation.isError && <Alert type="error" header="Invoke failed">{(mutation.error as Error).message}</Alert>}
        {mutation.data && <InvokeResult result={mutation.data} />}
      </SpaceBetween>
    }
    ```
