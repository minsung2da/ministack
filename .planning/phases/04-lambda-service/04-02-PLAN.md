    import { lambdaKeys } from './lambdaKeys'
    import type { LambdaFunctionSummary } from '../../../shared/types'

    export function useFunctions(marker: string | null = null, maxItems = 50) {
      return useQuery({
        queryKey: lambdaKeys.functions(marker),

          const sp: Record<string, string> = { MaxItems: String(maxItems) }
          if (marker) sp.Marker = marker
          return lambdaGet<ListFunctionsResponse>('/2015-03-31/functions', sp)
        },
        placeholderData: (prev) => prev,
      })
    }

    ```ts
    import { useMutation, useQueryClient } from '@tanstack/react-query'
    import { lambdaPost, lambdaPut, lambdaDelete } from './lambdaClient'
    import { lambdaKeys } from './lambdaKeys'
    import type { LambdaFunctionConfiguration, LambdaCodeSource } from '../../../shared/types'

    export type CreateFunctionInput = {
      FunctionName: string
      Runtime: string
      Role: string
      Handler: string
      Description?: string
      Timeout?: number
      MemorySize?: number
      Environment?: { Variables: Record<string, string> }
      Architectures?: ('x86_64' | 'arm64')[]
      Code: LambdaCodeSource
    }

    function buildCreateBody(input: CreateFunctionInput): Record<string, unknown> {
      const code = input.Code
      // Discriminated: emit exactly the keys the backend expects for this variant.
      // (RESEARCH §Backend REST Inventory item 2; backend line 654-670: ImageUri wins → PackageType=Image)
      let Code: Record<string, string>
      let PackageType: 'Zip' | 'Image' = 'Zip'
      if (code.kind === 'zip') {
        Code = { ZipFile: code.ZipFile }
      } else if (code.kind === 's3') {
        Code = { S3Bucket: code.S3Bucket, S3Key: code.S3Key }
      } else {
        Code = { ImageUri: code.ImageUri }
        PackageType = 'Image'
      }
      const { Code: _omit, ...rest } = input
      return { ...rest, Code, PackageType }
    }

    export function useCreateFunction() {
      const qc = useQueryClient()
      return useMutation({
        mutationFn: (input: CreateFunctionInput) =>
          lambdaPost<LambdaFunctionConfiguration>('/2015-03-31/functions', buildCreateBody(input)),
        onSuccess: () => { qc.invalidateQueries({ queryKey: lambdaKeys.all }) /* prefix-match covers all markers + all functions */ },
      })
    }
    // Note: lambdaKeys.all = ['lambda']; prefix-match invalidates functions(any marker) + function(any name) + triggers + functionUrl.
    // That's intentional after Create: the new function's detail isn't cached yet, and invalidating 'lambda' root is one call (Pitfall C-3 compliant — one invalidation per onSuccess).
    // CORRECTION: per RESEARCH line 719, use lambdaKeys.functions() (list only). Triggers/functionUrl stay valid for other functions. Use the narrower key:
    //   qc.invalidateQueries({ queryKey: lambdaKeys.functions() })  — prefix-matches all marker values.

    export function useDeleteFunction() {
      const qc = useQueryClient()
      return useMutation({
        mutationFn: (name: string) => lambdaDelete(`/2015-03-31/functions/${encodeURIComponent(name)}`),
        // NO Qualifier (Pitfall 9). lambdaDelete has no searchParams param.
        onSuccess: (_result, name) => {
          qc.removeQueries({ queryKey: lambdaKeys.function(name) })
          qc.invalidateQueries({ queryKey: lambdaKeys.functions() })
        },
      })
    }

    // UpdateConfiguration whitelist per RESEARCH §item 5
    const UPDATE_WHITELIST = [
      'Runtime', 'Handler', 'Description', 'Timeout', 'MemorySize', 'Role',
      'Environment', 'Layers', 'TracingConfig', 'DeadLetterConfig', 'KMSKeyArn',
      'EphemeralStorage', 'LoggingConfig', 'VpcConfig', 'Architectures',
      'FileSystemConfigs', 'ImageConfig',
    ] as const
    export type UpdateConfigurationInput = { name: string; patch: Partial<Record<typeof UPDATE_WHITELIST[number], unknown>> }

    export function useUpdateConfiguration() {
      const qc = useQueryClient()
      return useMutation({
        mutationFn: ({ name, patch }: UpdateConfigurationInput) =>
          lambdaPut<LambdaFunctionConfiguration>(`/2015-03-31/functions/${encodeURIComponent(name)}/configuration`, patch),
        onSuccess: (_result, { name }) => {
          // Pitfall C-3: single invalidation. Don't touch functions list — config fields we surface
          // in the list (runtime/handler/lastModified) will refetch via normal staleTime when list is revisited.
          qc.invalidateQueries({ queryKey: lambdaKeys.function(name) })
        },
      })
    }
    ```

<threat_model>
## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| React component → mutation hook | User-initiated Create/Delete/Update/Invoke cross here |
| User-supplied env var values → UpdateConfiguration body | Arbitrary strings, not validated client-side (D-05 plaintext) |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-4-02-01 | Tampering (JSON injection) | lambdaPost body serialization | mitigate | `JSON.stringify` on a structured object — no string concatenation. Backend parses with `json.loads`. |
| T-4-02-02 | Business Logic (V11 destructive action) | useDeleteFunction | accept | Type-to-confirm is the UI gate (Plan 03 DeleteFunctionModal). Mutation hook itself is un-guarded by design. |
| T-4-02-03 | Tampering (accidental version delete) | useDeleteFunction Qualifier | mitigate | lambdaDelete has no searchParams parameter (Pitfall 9); no path exists to send a Qualifier. Test asserts URL matches exactly `/functions/{name}`. |
| T-4-02-04 | Information Disclosure | UpdateConfiguration whitelist | accept | Server enforces whitelist at `_update_config` line 842-860; client type uses whitelist constant for compile-time guidance, not as a security boundary. |
| T-4-02-05 | Cache invalidation thrash (self-DoS) | multiple invalidateQueries per mutation | mitigate | Exactly one invalidateQueries (+ optional removeQueries) per onSuccess. Pitfall C-3 enforced by spy-based test. |
| T-4-02-06 | Stale cache leak | Invoke result cached across functions | mitigate | Pitfall 6: invoke NEVER invalidates, but it also never lives in a query — mutation.data is ephemeral. Test spies on invalidateQueries to confirm. |
</threat_model>
